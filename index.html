function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var clientsSheet = ss.getSheetByName("Clients"); var logsSheet = ss.getSheetByName("Logs"); var clientsData = clientsSheet ? clientsSheet.getDataRange().getValues() : []; var logsData = logsSheet ? logsSheet.getDataRange().getValues() : []; var clients = []; for(var i = 1; i < clientsData.length; i++) { if(clientsData[i][0]) { clients.push({ name: clientsData[i][0], type: clientsData[i][1], rep: clientsData[i][2] }); } } var logs = []; for(var j = 1; j < logsData.length; j++) { if(logsData[j][0]) { logs.push({ client: logsData[j][0], rep: logsData[j][1], details: logsData[j][2], isoDate: logsData[j][3], displayDate: logsData[j][4] }); } } return ContentService.createTextOutput(JSON.stringify({ clients: clients, logs: logs })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if(data.action === "addClient") { var sheet = ss.getSheetByName("Clients"); sheet.appendRow([data.client.name, data.client.type, data.client.rep]); } else if(data.action === "saveInventory") { var sheet = ss.getSheetByName("Logs"); sheet.appendRow([data.log.client, data.log.rep, data.log.details, data.log.isoDate, data.log.displayDate]); } return ContentService.createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); }
